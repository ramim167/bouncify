/*
function iSpecialKeyboard() is called whenver user hits special keys likefunction
keys, home, end, pg up, pg down, arraows etc. you have to use
appropriate constants to detect them. A list is:
GLUT_KEY_F1, GLUT_KEY_F2, GLUT_KEY_F3, GLUT_KEY_F4, GLUT_KEY_F5, GLUT_KEY_F6,
GLUT_KEY_F7, GLUT_KEY_F8, GLUT_KEY_F9, GLUT_KEY_F10, GLUT_KEY_F11,
GLUT_KEY_F12, GLUT_KEY_LEFT, GLUT_KEY_UP, GLUT_KEY_RIGHT, GLUT_KEY_DOWN,
GLUT_KEY_PAGE_UP, GLUT_KEY_PAGE_DOWN, GLUT_KEY_HOME, GLUT_KEY_END,
GLUT_KEY_INSERT */
void iSpecialKeyboard(unsigned char key)
{
    switch (key)
    {
    case GLUT_KEY_UP:
        ballY += 30;
        if (ballY+radius_ball>SCREEN_HEIGHT) ballY= SCREEN_HEIGHT -radius_ball;
        break;
    case GLUT_KEY_DOWN:
        ballY -= 30;
        if (ballY<210) ballY= 210;
        break;
    case GLUT_KEY_LEFT:
        ballX -= 30;
        if (ballX-radius_ball<0) ballX= radius_ball;
        break;
    case GLUT_KEY_RIGHT:
        ballX += 30;
        if (ballX+radius_ball>SCREEN_WIDTH-200) ballX= SCREEN_WIDTH -radius_ball-200;
        break;
    default:
        ballY=ballY-30;
        if (ballY<400) ballY = 400;
    }
}
